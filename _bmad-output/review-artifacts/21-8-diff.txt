+def _to_int(raw_value: Any) -> int:
+    """Convert a raw engagement count to an integer, defaulting to 0 on failure.
+
+    Handles human-readable suffixes (K, M, B, T) and numeric/empty values.
+    """
+    if raw_value is None or raw_value == "":
+        return 0
+    if isinstance(raw_value, bool):
+        return int(raw_value)
+    if isinstance(raw_value, (int, float)):
+        return int(raw_value)
+    if not isinstance(raw_value, str):
+        return 0
+
+    s = raw_value.strip().replace(",", "")
+    if not s:
+        return 0
+
+    m = re.fullmatch(r"([\d.]+)\s*([KkMmBbTt])?", s)
+    if m:
+        try:
+            num = float(m.group(1))
+        except ValueError:
+            return 0
+        suffix = m.group(2)
+        multiplier = 1.0
+        if suffix:
+            multipliers = {"K": 1e3, "M": 1e6, "B": 1e9, "T": 1e12}
+            multiplier = multipliers.get(suffix.upper(), 1.0)
+        return int(num * multiplier)
+
+    try:
+        return int(float(s))
+    except ValueError:
+        return 0

+def _parse_published_at(value: Any) -> datetime | None:
+    """Parse a published timestamp, returning None and logging on failure."""
+    if not value:
+        return None
+    if isinstance(value, datetime):
+        return value
+    if not isinstance(value, str):
+        return None
+
+    v = value.strip()
+    if v.endswith("Z"):
+        v = v[:-1] + "+00:00"
+
+    try:
+        return datetime.fromisoformat(v)
+    except ValueError:
+        pass
+
+    for fmt in (
+        "%a %b %d %H:%M:%S %z %Y",
+        "%a %b %d %H:%M:%S %Y",
+        "%Y-%m-%d %H:%M:%S%z",
+        "%Y-%m-%d %H:%M:%S",
+    ):
+        try:
+            return datetime.strptime(v, fmt)
+        except ValueError:
+            continue
+
+    logger.warning(
+        "Could not parse published timestamp %r; leaving published_at=None",
+        value,
+    )
+    return None
